-- 1. 필수 6대 정제 컬럼 빠른 조회
SELECT caseID, taskID, task_group, originator, seq, clean_timestamp
FROM repair_logs
ORDER BY caseID, clean_timestamp, seq, sub_seq;

-- 2. 케이스별 총 소요시간(Lead Time, 일) 집계
SELECT caseID,
       ROUND((JULIANDAY(MAX(clean_timestamp)) - JULIANDAY(MIN(clean_timestamp))), 2) AS lead_time_days
FROM repair_logs
GROUP BY caseID;
